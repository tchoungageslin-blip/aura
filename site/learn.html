2 · The tour2 · Le tour

fn square(x: i64) -> i64 { x * x }

fn might_fail(ok: bool) -> Result<i64, str> {
    if ok { Ok(42) } else { Err("nope") }
}

fn main() -> i64 {
    let name = "world"          // inferred, immutable
    let mut count = 0           // `mut` makes it assignable

    // `if` is an expression
    let label = if count > 0 { "yes" } else { "no" }
    while count < 10 { count = count + 1 }

    println("hello " + name)
    println(str_from_int(square(7)))
    match might_fail(true) {
        Ok(v)  => println(str_from_int(v)),
        Err(e) => eprintln(e),
    }
    return 0
}

Statements are separated by newlines; the last expression is the value of a block. Strings and numbers are different types — convert with str_from_int.Les instructions sont séparées par des retours à la ligne ; la dernière expression est la valeur d'un bloc. Chaînes et nombres sont des types distincts — convertis avec str_from_int.

fn main() -> i64 {
    let inp = read_stdin()
    let mut lines = 0
    let mut i = 0
    while i < inp.len {
        if str_get(inp, i) == 10 {   // byte 10 = '\n'
            lines = lines + 1
        }
        i = i + 1
    }
    println("lines: " + str_from_int(lines))
    0
}
